// getrings extracts the ring of the ideal and manufactures
// local copies of the coefficient field (kk), the ring (S, assumed 
// to be a polynomial ring), the dimension, and the ideal itself.

function getrings(Icenter)
    // Extract the polynomial ring containing the ideal
    S := Generic(Icenter); 
    
    // Check if S is a polynomial ring (multivariate or univariate)
    if not (Type(S) in {RngMPol, RngUPol}) then
        print "Sorry, expecting an ideal defined over a polynomial ring";
        return false, _, _, _; // Return false to indicate failure
    end if;
    
    kk := BaseRing(S);
    num_vars := Rank(S);
    n := num_vars - 1; 
    
    // 1. Create a new polynomial ring over kk with the same number of variables.
    // 2. Assign the string names z_0, ..., z_n so they print exactly like M2.
    names := ["z_" * IntegerToString(i) : i in [0..n]];
    S_new<[z]> := PolynomialRing(kk, num_vars);
    AssignNames(~S_new, names);
    
    // Map the ideal into the new ring
    // 'hom' creates a homomorphism sending the variables of S to the sequence z
    phi := hom< S -> S_new | z >;
    I_new := ideal< S_new | [phi(g) : g in Basis(Icenter)] >;
    
    // Return the items as multiple values (Magma's equivalent to an M2 sequence)
    return kk, S_new, n, I_new;
end function;

function fixideal(S, n, Icenter)
    // Extract a minimal set of generators (Magma's equivalent of M2's 'trim')
    // We assume Icenter is already an ideal of S (as prepared by getrings)
    center_gens := MinimalBasis(Icenter); 
    m := #center_gens;
    
    // Handle the zero ideal case explicitly
    if m eq 0 or (m eq 1 and center_gens[1] eq 0) then
        return 0, ideal< S | 0 >;
    end if;
    
    // Find the maximum total degree among all generators
    ma := Max([ TotalDegree(g) : g in center_gens ]);
    
    new_gens := [];
    
    for en in center_gens do
        cr := ma - TotalDegree(en);
        
        if cr eq 0 then
            // If it's already at max degree, just keep it
            Append(~new_gens, en);
        else
            // If lower degree, pad it by multiplying by each variable^cr
            // Note: Magma uses 1-based indexing, so variables are S.1 to S.(n+1)
            for j in [1..Rank(S)] do
                Append(~new_gens, en * (S.j)^cr);
            end for;
        end if;
    end for;
    
    // Create the new padded ideal
    tem := ideal< S | new_gens >;
    
    // Trim it down to a minimal basis before returning
    tem_trimmed := ideal< S | MinimalBasis(tem) >;
    
    return ma, tem_trimmed;
end function;

function blup(I)
    // Extract the ring and base field
    S := Generic(I);
    kk := BaseRing(S);
    
    // Get the minimal generators of the ideal
    I_gens := MinimalBasis(I);
    m := #I_gens;
    S_vars := Rank(S);
    
    // Create the new polynomial ring R = kk[gens S, t_0 .. t_(m-1)]
    // It needs Rank(S) + m total variables
    total_vars := S_vars + m;
    R := PolynomialRing(kk, total_vars);
    
    // Assign variable names dynamically so they match M2
    // 1. Grab the existing names from S (e.g., z_0, z_1) using Sprint
    // 2. Append the new t_i variables
    old_names := [ Sprint(S.i) : i in [1..S_vars] ]; 
    t_names := [ "t_" * IntegerToString(i) : i in [0..m-1] ];
    AssignNames(~R, old_names cat t_names);
    
    // Map the old variables into the new ring R
    phi := hom< S -> R | [ R.i : i in [1..S_vars] ] >;
    
    // Map the generators of I into R
    II_gens := [ phi(g) : g in I_gens ];
    
    // Handle the trivial/principal cases
    if m eq 0 then
        return ideal< R | 0 >;
    elif m eq 1 then
        J := ideal< R | 0 >;
    else
        // Construct the ideal J generated by the 2x2 minors (II_i * t_j - II_j * t_i)
        // Note: Magma uses 1-based indexing! 
        // t_i corresponds to R.(S_vars + i)
        rels := [];
        for i in [1 .. m-1] do
            for j in [i+1 .. m] do
                // Cross-multiply to get the syzygy relations
                rel := II_gens[i] * R.(S_vars + j) - II_gens[j] * R.(S_vars + i);
                Append(~rels, rel);
            end for;
        end for;
        J := ideal< R | rels >;
    end if;
    
    // If the ideal was generated by 0, safely return the 0 ideal to avoid saturation errors
    if m eq 1 and II_gens[1] eq 0 then
        return J;
    end if;
    
    // Compute the saturation of J with respect to the first generator (II_0)
    Rees_ideal := Saturation(J, ideal< R | II_gens[1] >);
    
    // Magma ties ideals to their rings implicitly, but returning both is a 
    // good practice so you have easy access to R in the global scope.
    return R, Rees_ideal;
end function;

// Helper function to simulate M2's multidegree coefficients
// 1. The BiDegree function we just debugged
function BiDegreeCoefficients(Ublowup, U, m, S_vars, n)
    degrees := [];
    // Helper to generate non-zero random coefficients
    rand_coeff := func< | Random([1..10]) >; 
    
    for i in [0..n] do 
        num_t_cuts := i;
        num_z_cuts := n - i;
        
        // Boundary check: if we exceed variables, the intersection is empty
        if num_t_cuts ge m or num_z_cuts ge S_vars then
            Append(~degrees, 0);
            continue;
        end if;
        
        J := Ublowup;
        
        // Homogeneous 't' cuts
        for cut in [1..num_t_cuts] do
            form := &+[ rand_coeff() * U.j : j in [1..m] ];
            J := J + ideal< U | form >;
        end for;
        
        // Homogeneous 'z' cuts
        for cut in [1..num_z_cuts] do
            form := &+[ rand_coeff() * U.j : j in [m+1 .. m+S_vars] ];
            J := J + ideal< U | form >;
        end for;
        
        // Affine patch 't' (dodge t=0)
        form_t_aff := &+[ rand_coeff() * U.j : j in [1..m] ] - 1;
        J := J + ideal< U | form_t_aff >;
        
        // Affine patch 'z' (dodge z=0)
        form_z_aff := &+[ rand_coeff() * U.j : j in [m+1 .. m+S_vars] ] - 1;
        J := J + ideal< U | form_z_aff >;
        
        // Compute the degree of the 0-dimensional affine intersection
        if Dimension(J) eq 0 then
            deg := QuotientDimension(J);
            Append(~degrees, deg);
        else
            Append(~degrees, 0); // Failsafe if dimension didn't drop
        end if;
    end for;

    return degrees;
end function;


// 2. The Main PreSegre Translation
function presegre(S, I, n)
    k := BaseRing(S);
    gens_I := Basis(I); 
    m := #gens_I;
    S_vars := Rank(S);
    
    // 1. Create a "Big" ring P that contains: y, t_1..t_m, z_1..z_s
    // P.1 is y. P.2 to P.(m+1) are the t's. The rest are the z's.
    P := PolynomialRing(k, 1 + m + S_vars);
    y := P.1;
    
    // 2. Map the base ring S into P (S.i maps to the 'z' variables in P)
    phi_S_P := hom< S -> P | [ P.(1 + m + i) : i in [1..S_vars] ] >;
    
    // 3. Define the equations of the Rees algebra: t_i - f_i * y
    eqns := [ P.(1 + i) - phi_S_P(gens_I[i]) * y : i in [1..m] ];
    J_big := ideal< P | eqns >;
    
    // 4. Eliminate 'y'. In Magma, we specify the variables we want to KEEP.
    keep_vars := { P.i : i in [2 .. 1 + m + S_vars] };
    J_elim := EliminationIdeal(J_big, keep_vars);
    
    // 5. Create our target ambient ring U = k[t_1..t_m, z_1..z_s]
    U := PolynomialRing(k, m + S_vars);
    
    // 6. Map the eliminated ideal from P back down to U
    // y maps to 0 (it's gone anyway), P.i maps to U.(i-1)
    phi_P_U := hom< P -> U | [0] cat [ U.i : i in [1 .. m + S_vars] ] >;
    
    // This perfectly replicates Kernel(phi)!
    Ublowup := ideal< U | [ phi_P_U(g) : g in Basis(J_elim) ] >;
    
    // Handle the trivial case
    if Ublowup eq ideal< U | 1 > then
        return [ 0 : i in [0..n] ];
    end if;
    
    // Calculate and return the bi-degree sequence
    return BiDegreeCoefficients(Ublowup, U, m, S_vars, n);
end function;
function segre(S, I, n)
    // 1. Get the bi-degree sequence from our PreSegre function
    tem := presegre(S, I, n);
    
    // 2. Find 'ma', the degree of the generators of the center ideal.
    // We assume I is generated by forms of the same degree.
    gens_I := Basis(I);
    ma := TotalDegree(gens_I[1]);
    
    // 3. Create the intersection ring intringPn = Z[H] / (H^(n+1))
    Z_H<H_var> := PolynomialRing(Integers());
    intringPn<H> := quo< Z_H | H_var^(n+1) >;
    
    // 4. Compute 'poly'
    // Magma sequences are 1-indexed, so we use tem[s+1] to match M2's tem#s
    poly := &+[ tem[s+1] * H^s * (1 + ma*H)^(n-s) : s in [0..n] ];
    
    // 5. Compute the Segre class
    segre_sum := &+[ Binomial(n+i, i) * (-ma*H)^i : i in [0..n] ];
    segreclass := intringPn!1 - poly * segre_sum;
    
    // 6. Compute the Fulton class
    fultonclass := (intringPn!1 + H)^(n+1) * segreclass;
    
    // Output the results directly to the terminal
    
    return segreclass, fultonclass;
end function;
function AnalyzeRationalMap(S, I, n)
// ==========================================
    // CROSSCHECKS & INPUT VALIDATION
    // ==========================================
    gens_I := Basis(I);
    // 1. Check if we have exactly n+1 generators
    if #gens_I ne n + 1 then
        error Sprintf("Input Error: The ideal must be generated by exactly %o polynomials (n+1) for a map from P^%o. You provided %o.", n + 1, n, #gens_I);
    end if;
    
    // 2. Check if all generators are homogeneous and of the same degree
    ma := TotalDegree(gens_I[1]); // Expected degree d
    
    for f in gens_I do
        if not IsHomogeneous(f) then
            error "Input Error: All defining polynomials must be strictly homogeneous.";
        end if;
        
        if TotalDegree(f) ne ma then
            error Sprintf("Input Error: All polynomials must have the exact same degree.");
        end if;
    end for;
    // 1. Compute the multidegrees only once
    multidegs := presegre(S, I, n);
    
    // 2. The degree of the rational map is the last entry in the sequence
    // Since Magma is 1-indexed and we go from 0 to n, the last index is n+1
    map_degree := multidegs[n+1];
    is_birational := (map_degree eq 1);
    
    // 3. Compute the Segre class using the multidegrees
    gens_I := Basis(I);
    ma := TotalDegree(gens_I[1]); // Degree d of the n+1 forms
    
    Z_H<H_var> := PolynomialRing(Integers());
    intringPn<H> := quo< Z_H | H_var^(n+1) >;
    
    poly := &+[ multidegs[s+1] * H^s * (1 + ma*H)^(n-s) : s in [0..n] ];
    segre_sum := &+[ Binomial(n+i, i) * (-ma*H)^i : i in [0..n] ];
    
    segreclass := intringPn!1 - poly * segre_sum;
    
    // 4. Print a clean, formatted output summary
    print "================================================";
    printf "Multidegrees   : %o\n", multidegs;
    printf "Map Degree     : %o\n", map_degree;
    
    if is_birational then
        print "Birationality  : TRUE (The map is birational!)";
    else
        print "Birationality  : FALSE (The map is not birational)";
    end if;
    
    printf "Segre class    : %o\n", segreclass;
	// Lift the segreclass back to Z[H] to extract coefficients
    lifted_segre := Z_H ! segreclass; 
    
    for i in [0..n] do
        coeff := Coefficient(lifted_segre, i);
        if i eq 0 then
            printf "H^%o = %o;\n", n, coeff+1;
        elif i eq n then
            printf "E^%o = %o.\n", n, coeff;
        else
            printf "H^%o * E^%o = %o;\n", n-i, i, coeff;
        end if;
    end for;
    print "================================================";
    
    // Return all three elements as a tuple for further use
    return multidegs, segreclass, is_birational;
end function;

function BaseLocusAnalysis(Z)
	print "Base locus analysis:";
	print "Dimension:", Dimension(Z);
	print "Degree:", Degree(Z);
	Comps := IrreducibleComponents(Z);
	print "Number of Irreducible Components equal to ",#Comps; 
	print "Base Locus Components:";
	for C in Comps do
    		print "Dimension:", Dimension(C), "Degree:", Degree(C);
    		C;
    		if Dimension(C) eq 0 then
        		Points(C);
    		end if;
	end for;
	return 0;
end function;

// Compute multidegree of a map from P^4 to P^4


FF := Rationals();
n := 4;
P<[u]> := ProjectiveSpace(FF, n);

// Define the map components
beta := [];
beta[1] := u[2]*u[5]-u[3]*u[4];
beta[2] := u[2]*u[4]-u[3]^2+u[3]*u[5]-u[4]^2;
beta[3] := u[1]*u[3];
beta[4] := u[1]*u[5];
beta[5] := u[1]*(u[2]-u[4]);
R := CoordinateRing(P);
beta_poly := [R!b : b in beta];
base_ideal := ideal<R | beta_poly>;
__, S_new, n, I_new := getrings(base_ideal);
BaseLocusAnalysis(Scheme(P,base_ideal));
multidegs, segre_c, is_bir := AnalyzeRationalMap(S_new, I_new, n);
